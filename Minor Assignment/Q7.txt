# Q7 - HAProxy load balancing with two web backends and an API

mkdir -p haproxy-assignment
cd haproxy-assignment

sudo docker network create proxy-net
Output=>
<network-id>

cat > haproxy.cfg <<'EOF'
global
    log stdout format raw local0

defaults
    log global
    mode http
    timeout connect 5s
    timeout client 30s
    timeout server 30s

frontend public
    bind *:80
    acl is_api path_beg /api
    use_backend api_service if is_api
    default_backend web_services

backend web_services
    balance roundrobin
    server web1 web1:80 check
    server web2 web2:80 check

backend api_service
    server api api:80 check
EOF

cat haproxy.cfg
Output=>
[Displays the HAProxy configuration above.]

sudo docker run -d --name web1 --network proxy-net traefik/whoami
Output=>
<container-id>

sudo docker run -d --name web2 --network proxy-net traefik/whoami
Output=>
<container-id>

sudo docker run -d --name api --network proxy-net traefik/whoami
Output=>
<container-id>

sudo docker run -d --name haproxy --network proxy-net -p 8088:80 -v "$PWD/haproxy.cfg:/usr/local/etc/haproxy/haproxy.cfg:ro" haproxy:alpine
Output=>
<container-id>

sudo docker ps
Output=>

<container-id> traefik/whoami     Up                              api

curl -s http://localhost:8088/

sudo docker network rm proxy-net

proxy-net

# Notes for written discussion:
# Backends share proxy-net and are reached by Docker service/container names.
# Only HAProxy publishes a host port; backend/API ports are not published.
# HAProxy uses a static config file. Traefik commonly discovers containers via labels.
# Docker socket access gives a proxy control-plane visibility and should be restricted.
# TLS can terminate at the proxy; Traefik can automate certificates with an ACME provider.
