# Q7 - Run Nginx and PHP-FPM as separate Docker Compose services

# Step 1: Create the project structure
mkdir -p ~/CFD-Assignment-4/Q7/nginx
cd ~/CFD-Assignment-4/Q7

# Step 2: Create a PHP page that adds two numbers
cat > index.php <<'PHP'
<?php
$a = filter_input(INPUT_GET, 'a', FILTER_VALIDATE_FLOAT);
$b = filter_input(INPUT_GET, 'b', FILTER_VALIDATE_FLOAT);

if ($a === false || $b === false || $a === null || $b === null) {
    $a = 0;
    $b = 0;
}
$sum = $a + $b;
?>
<!doctype html>
<html lang="en">
<head><meta charset="utf-8"><title>PHP Sum Calculator</title></head>
<body>
  <h1>PHP Sum Calculator</h1>
  <form method="get">
    <label>First number: <input type="number" step="any" name="a" required></label>
    <label>Second number: <input type="number" step="any" name="b" required></label>
    <button type="submit">Calculate</button>
  </form>
  <p>Sum: <?= htmlspecialchars((string)$sum, ENT_QUOTES, 'UTF-8') ?></p>
</body>
</html>
PHP

# Step 3: Create the Nginx configuration
cat > nginx/default.conf <<'NGINX'
server {
    listen 80;
    server_name localhost;
    root /var/www/html;
    index index.php index.html;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        try_files $uri =404;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_pass php-fpm:9000;
    }
}
NGINX

# Step 4: Create the Docker Compose file
cat > compose.yaml <<'YAML'
services:
  nginx:
    image: nginx:1.27-alpine
    container_name: assignment4-nginx
    ports:
      - "8081:80"
    volumes:
      - ./index.php:/var/www/html/index.php:ro
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - php-fpm
    restart: unless-stopped

  php-fpm:
    image: php:8.3-fpm-alpine
    container_name: assignment4-php-fpm
    volumes:
      - ./index.php:/var/www/html/index.php:ro
    expose:
      - "9000"
    restart: unless-stopped
YAML

# Step 5: Start the services and check their status
sudo docker compose up -d
sudo docker compose ps

# Step 6: Test through HTTP
curl "http://localhost:8081/?a=12&b=30"

Output=>
The returned HTML should include 'Sum: 42'.

Browser:
http://localhost:8081/?a=12&b=30

# Step 7: Confirm only Nginx publishes a host port
sudo docker compose ps
sudo docker port assignment4-nginx
sudo docker port assignment4-php-fpm

Output=>
Nginx shows host port 8081 mapped to container port 80. PHP-FPM has no host-published port; it is reachable by Nginx over the Compose network.

# Step 8: Stop PHP-FPM and observe application behavior
sudo docker compose stop php-fpm
curl -i "http://localhost:8081/?a=12&b=30"

Output=>
The request should fail with an Nginx gateway error (commonly HTTP 502 Bad Gateway) because Nginx cannot connect to the stopped PHP-FPM service.

# Step 9: Restart PHP-FPM and verify recovery
sudo docker compose start php-fpm
curl "http://localhost:8081/?a=12&b=30"

Notes:
- Nginx serves HTTP and forwards PHP requests to the Compose service name 'php-fpm' on port 9000.
- 'expose' documents the internal PHP-FPM port; it does not publish the port on the host.
- Only Nginx uses the 'ports' setting.
- The containers share the PHP file through a read-only bind mount.
