# Q9 - Python Flask app, standard vs slim image, resource limits

mkdir -p python-container-app
cd python-container-app

cat > app.py <<'EOF'
from flask import Flask, jsonify
app = Flask(__name__)

@app.get("/")
def home():
    return jsonify(application="Docker demo", status="running")

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
EOF

cat > requirements.txt <<'EOF'
Flask==3.1.1
EOF

cat > Dockerfile.standard <<'EOF'
FROM python:3.12
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app.py .
RUN useradd --create-home appuser
USER appuser
EXPOSE 5000
CMD ["python", "app.py"]
EOF

cat > Dockerfile.slim <<'EOF'
FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app.py .
RUN useradd --create-home appuser
USER appuser
EXPOSE 5000
CMD ["python", "app.py"]
EOF

sudo docker build -t flask-demo:standard -f Dockerfile.standard .
Output=>
Successfully tagged flask-demo:standard

sudo docker build -t flask-demo:slim -f Dockerfile.slim .
Output=>
Successfully tagged flask-demo:slim

sudo docker image ls flask-demo
Output=>

flask-demo    slim      <image-id>     <time>         <slim-size>
flask-demo    standard  <image-id>     <time>         <standard-size>
(Actual sizes vary; compare the SIZE column.)

sudo docker run -d --name flask-standard -p 5000:5000 --cpus="0.5" --memory="128m" flask-demo:standard
