cat > app/app.py <<'EOF'
import os, time, psycopg
from flask import Flask, jsonify, request

app = Flask(__name__)
DB_HOST = os.getenv("DB_HOST", "db")
DB_NAME = os.environ["POSTGRES_DB"]
DB_USER = os.environ["POSTGRES_USER"]
DB_PASSWORD = os.environ["POSTGRES_PASSWORD"]
APP_PORT = int(os.getenv("APP_PORT", "5000"))

def connect_db():
    return psycopg.connect(host=DB_HOST, dbname=DB_NAME, user=DB_USER,
                           password=DB_PASSWORD, connect_timeout=3)

def initialize_database():
    with connect_db() as conn:
        with conn.cursor() as cur:
            cur.execute("CREATE TABLE IF NOT EXISTS messages (id SERIAL PRIMARY KEY, content TEXT NOT NULL)")

def wait_for_database():
    for attempt in range(20):
        try:
            initialize_database()
            app.logger.info("Database is ready.")
            return
        except psycopg.OperationalError:
            app.logger.info("Waiting for database: attempt %s/20", attempt + 1)
            time.sleep(2)
    raise RuntimeError("Database did not become available in time.")

@app.get("/health")
def health():
    return jsonify({"status": "healthy"})

@app.get("/api/messages")
def get_messages():
    with connect_db() as conn:
        with conn.cursor() as cur:
            cur.execute("SELECT id, content FROM messages ORDER BY id")
            rows = cur.fetchall()
    return jsonify([{"id": row[0], "content": row[1]} for row in rows])

@app.post("/api/messages")
def create_message():
    data = request.get_json(silent=True) or {}
    content = data.get("content")
    if not isinstance(content, str) or not content.strip():
        return jsonify({"error": "content is required"}), 400
    with connect_db() as conn:
        with conn.cursor() as cur:
            cur.execute("INSERT INTO messages (content) VALUES (%s) RETURNING id", (content.strip(),))
            message_id = cur.fetchone()[0]
    return jsonify({"id": message_id, "content": content.strip()}), 201

if __name__ == "__main__":
    wait_for_database()
    app.run(host="0.0.0.0", port=APP_PORT)
EOF

cat > app/Dockerfile <<'EOF'
FROM python:3.12-slim
WORKDIR /app
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt
COPY app.py .
EXPOSE 5000
CMD ["python", "app.py"]
EOF

cat > web/nginx.conf <<'EOF'
server {
    listen 80;
    server_name _;
    location / {
        default_type text/html;
        return 200 '<!doctype html><html><head><title>Three-Service Application</title></head><body><h1>Three-Service Docker Application</h1><p>Web: Nginx; App: Flask; Database: PostgreSQL</p><button onclick="loadMessages()">Load Messages</button><button onclick="addMessage()">Add Message</button><pre id="output"></pre><script>async function loadMessages(){let r=await fetch("/api/messages");document.getElementById("output").textContent=JSON.stringify(await r.json(),null,2)}async function addMessage(){let r=await fetch("/api/messages",{method:"POST",headers:{"Content-Type":"application/json"},body:JSON.stringify({content:"Hello from Docker Compose"})});document.getElementById("output").textContent=JSON.stringify(await r.json(),null,2)}</script></body></html>';
    }
    location /api/ {
        proxy_pass http://app:5000/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
    location /health {
        proxy_pass http://app:5000/health;
        proxy_set_header Host $host;
    }
}
EOF

cat > compose.yml <<'EOF'
services:
  web:
    image: nginx:alpine
    ports:
      - "8080:80"
    volumes:
      - ./web/nginx.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - app
    networks:
      - app-network
    restart: unless-stopped

  app:
    build:
      context: ./app
    environment:
      DB_HOST: db
      POSTGRES_DB: ${POSTGRES_DB:?POSTGRES_DB is required}
      POSTGRES_USER: ${POSTGRES_USER:?POSTGRES_USER is required}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?POSTGRES_PASSWORD is required}
      APP_PORT: ${APP_PORT:-5000}
    expose:
      - "${APP_PORT:-5000}"
    depends_on:
      db:
        condition: service_healthy
    networks:
      - app-network
    restart: unless-stopped

  db:
    image: postgres:17
    environment:
      POSTGRES_DB: ${POSTGRES_DB:?POSTGRES_DB is required}
      POSTGRES_USER: ${POSTGRES_USER:?POSTGRES_USER is required}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:?POSTGRES_PASSWORD is required}
    volumes:
      - db-data:/var/lib/postgresql/data
    networks:
      - app-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 3s
      retries: 10
      start_period: 5s
    restart: unless-stopped

volumes:
  db-data:

networks:
  app-network:
    driver: bridge
EOF

Validate, build and start:
sudo docker compose config
sudo docker compose up -d --build
sudo docker compose ps

Test:
Open http://localhost:8080
curl http://localhost:8080/health
curl http://localhost:8080/api/messages

Insert a message:
curl -X POST http://localhost:8080/api/messages -H "Content-Type: application/json" -d '{"content":"First message from Docker Compose"}'
curl http://localhost:8080/api/messages

Inspect:
sudo docker compose ps -a
sudo docker compose logs web
sudo docker compose logs app
sudo docker compose logs db
sudo docker compose top
sudo docker network ls
sudo docker volume ls

Inspect network/volume names (Compose project name may vary):
sudo docker network inspect q4-three-service_app-network
sudo docker volume inspect q4-three-service_db-data

Stop all services and then start existing containers:
sudo docker compose stop
sudo docker compose start

Stop or start only application:
sudo docker compose stop app
sudo docker compose start app

Recreate only app, without starting/recreating dependencies:
sudo docker compose up -d --no-deps --force-recreate app

Remove containers and network, retaining named volume:
sudo docker compose down

Remove containers, network and named volume/data (destructive):
sudo docker compose down -v
